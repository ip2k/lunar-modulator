def ancestor(p):
    for parent in [p.resolve(), *p.resolve().parents]:
        if (parent/'idVendor').exists():
            return parent
    return None


def node_record(node, sysfs, character):
    info = node.stat()
    valid = stat.S_ISCHR(info.st_mode) if character else stat.S_ISBLK(info.st_mode)
    dev = f'{os.major(info.st_rdev)}:{os.minor(info.st_rdev)}'
    if not valid or dev != (sysfs/'dev').read_text().strip():
        raise ValueError('Actual device node type/number differs from bound sysfs')
    return {'node': str(node), 'dev': dev}


def exclusive(records, log):
    mounted = {line.split()[2] for line in Path('/proc/self/mountinfo').read_text().splitlines()}
    if any(r['dev'] in mounted for r in records):
        raise ValueError('A reviewed device is mounted')
    check = subprocess.run(['fuser', '-s', *(r['node'] for r in records)], capture_output=True, timeout=5)
    log('exclusive_access_check', nodes=records, returncode=check.returncode,
        stdout=check.stdout.decode(errors='replace'), stderr=check.stderr.decode(errors='replace'))
    if check.returncode != 1 or check.stdout or check.stderr:
        raise ValueError('Cannot establish exclusive device access')


def same_uboot(expected):
    devices = enumerate_usb()
    found = [d for d in devices if (d['vid'], d['pid']) == ('4c4a', '8057')]
    if len(found) != 1 or any(found[0][k] != expected[k] for k in ('path', 'busnum', 'devnum', 'vid', 'pid', 'product')):
        raise ValueError('Current UBOOT differs from fresh recorded enumeration')
    if any((d['vid'], d['pid']) == ('4c4a', 'c755') for d in devices):
        raise ValueError('Unexpected normal FM-1 also connected')


def ready_sg(expected, log):
    usb = Path('/sys/bus/usb/devices/3-2').resolve()
    end = time.monotonic() + 5
    while True:
        same_uboot(expected)
        sg = [p for p in Path('/sys/class/scsi_generic').glob('sg*') if ancestor(p/'device') == usb]
        block = [p for p in Path('/sys/class/block').glob('*') if ancestor(p) == usb]
        if len(sg) > 1 or any(p.name != 'sg0' for p in sg) or any(p.name != 'sda' for p in block):
            raise ValueError('Reviewed SG/block node set changed')
        if len(sg) == 1 and len(block) == 1 and all((Path('/dev')/p.name).exists() for p in [*sg, *block]):
            break
        if time.monotonic() >= end:
            log('device_nodes_not_ready', sg=[p.name for p in sg], block=[p.name for p in block])
            raise ValueError('Exact SG/block device nodes not ready; stop')
        time.sleep(0.05)  # Bounded enumeration readiness polling; no command retry.
    records = [node_record(Path('/dev')/sg[0].name, sg[0], True),
               node_record(Path('/dev')/block[0].name, block[0], False)]
    exclusive(records, log)
    same_uboot(expected)
    return records[0]['node']
