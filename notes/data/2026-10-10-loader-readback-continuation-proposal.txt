def ancestor(p):
    for parent in [p.resolve(), *p.resolve().parents]:
        if (parent/'idVendor').exists():
            return parent
    return None


def bound_nodes(expected, log):
    devices = enumerate_usb()
    found = [d for d in devices if (d['vid'], d['pid']) == ('4c4a', '8057')]
    if len(found) != 1 or any(found[0][k] != expected[k] for k in ('path', 'busnum', 'devnum', 'vid', 'pid', 'product')):
        raise ValueError('Current UBOOT differs from the exact prior enumeration')
    if any((d['vid'], d['pid']) == ('4c4a', 'c755') for d in devices):
        raise ValueError('Unexpected normal FM-1 also connected')
    usb = Path('/sys/bus/usb/devices/3-2').resolve()
    sg = [p for p in Path('/sys/class/scsi_generic').glob('sg*') if ancestor(p/'device') == usb]
    block = [p for p in Path('/sys/class/block').glob('*') if ancestor(p) == usb]
    if len(sg) != 1 or sg[0].name != 'sg0' or sorted(p.name for p in block) != ['sda']:
        raise ValueError('Exact reviewed SG/block node set changed')
    records = []
    for p in [*sg, *block]:
        node = Path('/dev')/p.name
        info = node.stat()  # Actual node existence, not just sysfs presence.
        valid_type = stat.S_ISCHR(info.st_mode) if p in sg else stat.S_ISBLK(info.st_mode)
        dev = f'{os.major(info.st_rdev)}:{os.minor(info.st_rdev)}'
        if not valid_type or dev != (p/'dev').read_text().strip():
            raise ValueError('Device node type/number does not match USB-bound sysfs')
        records.append({'node': str(node), 'dev': dev})
    mounted = {line.split()[2] for line in Path('/proc/self/mountinfo').read_text().splitlines()}
    if any(r['dev'] in mounted for r in records[1:]):
        raise ValueError('UBOOT block device is mounted')
    check = subprocess.run(['fuser', '-s', *(r['node'] for r in records)], capture_output=True, timeout=5)
    log('exclusive_access_check', nodes=records, returncode=check.returncode,
        stdout=check.stdout.decode(errors='replace'), stderr=check.stderr.decode(errors='replace'), usb=found[0])
    if check.returncode != 1 or check.stdout or check.stderr:
        raise ValueError('Cannot establish exclusive device access')
    return records[0]['node']
